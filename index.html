create or replace function public.claim_locker( p_qr_token uuid, p_employee_id text, p_name text ) returns json language plpgsql security definer as $$ declare v_locker lockers%rowtype; v_employee employees%rowtype; begin select * into v_locker from lockers where qr_token = p_qr_token for update; if not found then return json_build_object('success', false, 'error', 'Skab ikke fundet'); end if; if v_locker.status <> 'available' then return json_build_object('success', false, 'error', 'Skabet er allerede optaget'); end if; insert into employees (employee_id, name, active) values (p_employee_id, p_name, true) on conflict (employee_id) do update set name = excluded.name returning * into v_employee; update lockers set status = 'occupied', employee_id = v_employee.id, assigned_at = now(), updated_at = now() where id = v_locker.id; insert into locker_history ( locker_id, employee_id, action, created_at ) values ( v_locker.id, v_employee.id, 'assigned', now() ); return json_build_object( 'success', true, 'locker_number', v_locker.locker_number ); end; $$; grant execute on function public.claim_locker(uuid, text, text) to anon; grant execute on function public.claim_locker(uuid, text, text) to authenticated;
